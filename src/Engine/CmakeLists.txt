# ------------------------------------------------------------------------------
# Shaders HLSL -> SPIR-V (dxc du SDK Vulkan LunarG), déposés à côté des
# exécutables. resolveEngineFile("shaders", ...) les trouve via platformExeDir().
# ------------------------------------------------------------------------------
if(WIN32)
  set(_dxc_names dxc.exe)
  file(GLOB _sdk_dxc "C:/VulkanSDK/*/Bin/dxc.exe")
else()
  set(_dxc_names dxc)
  file(GLOB _sdk_dxc "$ENV{HOME}/VulkanSDK/*/macOS/bin/dxc")
endif()
if(DEFINED ENV{VULKAN_SDK})
  find_program(DXC_EXE NAMES ${_dxc_names} PATHS "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin"
               NO_DEFAULT_PATH)
endif()
if(NOT DXC_EXE)
  list(SORT _sdk_dxc COMPARE NATURAL)
  list(POP_BACK _sdk_dxc DXC_EXE)
endif()
if(NOT DXC_EXE)
  message(FATAL_ERROR "dxc introuvable — installer le SDK Vulkan LunarG")
endif()

# Le même SDK fournit l'API runtime (hot reload) : headers dxc/dxcapi.h
# (VulkanShaderCompiler) et libdxcompiler (chargée à l'exécution ;
# BATAP_DXC_LIB_DIR sert de chemin de secours quand DYLD/PATH ne la couvre pas).
get_filename_component(_dxc_bin_dir "${DXC_EXE}" DIRECTORY)
get_filename_component(_dxc_sdk_root "${_dxc_bin_dir}" DIRECTORY)
if(EXISTS "${_dxc_sdk_root}/include")
  target_include_directories(Batap_Engine SYSTEM PRIVATE "${_dxc_sdk_root}/include")
else()
  target_include_directories(Batap_Engine SYSTEM PRIVATE "${_dxc_sdk_root}/Include")
endif()
if(EXISTS "${_dxc_sdk_root}/lib")
  target_compile_definitions(Batap_Engine PRIVATE BATAP_DXC_LIB_DIR="${_dxc_sdk_root}/lib")
else()
  target_compile_definitions(Batap_Engine PRIVATE BATAP_DXC_LIB_DIR="${_dxc_bin_dir}")
endif()

set(engine_shader_out "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders")
file(MAKE_DIRECTORY "${engine_shader_out}")

function(batap_compile_shader src out target)
  add_custom_command(
    OUTPUT "${engine_shader_out}/${out}"
    COMMAND "${DXC_EXE}" -T ${target} -E main
            -spirv -fspv-target-env=vulkan1.3
            "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${src}"
            -Fo "${engine_shader_out}/${out}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${src}"
    COMMENT "dxc ${src} -> ${out}"
  )
endfunction()

batap_compile_shader(VertexShader.hlsl VertexShader.spv vs_6_6)
batap_compile_shader(PixelShader.hlsl  PixelShader.spv  ps_6_6)
batap_compile_shader(SkyVS.hlsl        SkyVS.spv        vs_6_6)
batap_compile_shader(SkyPS.hlsl        SkyPS.spv        ps_6_6)

add_custom_target(Batap_Shaders DEPENDS
  "${engine_shader_out}/VertexShader.spv"
  "${engine_shader_out}/PixelShader.spv"
  "${engine_shader_out}/SkyVS.spv"
  "${engine_shader_out}/SkyPS.spv")
add_dependencies(Batap_Engine Batap_Shaders)

# Hot reload sous Windows : dxcompiler.dll à côté des exes (pas de dxil.dll —
# la sortie SPIR-V n'est pas signée, contrairement au DXIL)

  file(COPY "${_dxc_bin_dir}/dxcompiler.dll"
       DESTINATION "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
endif()
